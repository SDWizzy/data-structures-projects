\section*{Instructions to Run}
\begin{enumerate}
    \item Open a terminal and navigate to the repository's root directory, the directory containing the \texttt{Project 1} folder.
    \item Enter the project directory:
\begin{verbatim}
cd "Project 1"
\end{verbatim}
    \item Configure the project with CMake:
\begin{verbatim}
cmake -S . -B build
\end{verbatim}
    \item Build the program:
\begin{verbatim}
cmake --build build
\end{verbatim}
    \item Run the Hat Simulator:
\begin{verbatim}
./build/data_structures_projects
\end{verbatim}
\end{enumerate}
